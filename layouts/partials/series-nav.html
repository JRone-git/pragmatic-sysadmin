{{/* Series navigation — "Part N of M" box for multi-part build series. Renders only when the post carries `series:` in its frontmatter. Two sources are merged: 1. Written parts — every post in the same series, ordered by the per-post `series_order` frontmatter field. 2. Planned parts — the `series_planned` list on the current post. This lets a series advertise parts that don't exist yet, so a reader landing on Part 1 can see the shape of the whole thing. Frontmatter contract: series: "Second Life" # taxonomy term (required) series_order: 1 # 1-based position among written parts series_title: "The €30 Prototype" # optional short title for the nav list series_planned: ["The €30 Prototype", "The €56 Wall Mount", ...] # optional: full ladder, written or not */}} {{- $term := .Params.series -}} {{- if $term -}} {{- $written := where (where site.RegularPages "Type" "in" site.Params.mainSections) "Params.series" $term -}} {{- if gt (len $written) 0 -}} {{- $sorted := sort $written "Params.series_order" "asc" -}} {{- $planned := .Params.series_planned -}} {{- $total := len $sorted -}} {{- with $planned -}}{{- $total = len . -}}{{- end -}} {{- $current := . -}} {{- $idx := 0 -}} {{- range $i, $p := $sorted -}} {{- if eq $p.RelPermalink $current.RelPermalink -}}{{- $idx = add $i 1 -}}{{- end -}} {{- end -}} {{- $seriesPage := site.GetPage (printf "/series/%s" (urlize $term)) -}}
Series {{ $term }} {{- if $idx }} Part {{ $idx }} of {{ $total }} {{- end }}
    {{- if $planned }} {{/* Full ladder from the plan — written parts link, planned ones don't. */}} {{- range $i, $title := $planned }} {{- $n := add $i 1 }} {{- $match := index (where $sorted "Params.series_title" $title) 0 -}} {{- if $match }} {{- $isCurrent := eq $match.RelPermalink $current.RelPermalink }}
  1. {{ $n }} {{ $title }} {{- if $isCurrent }}you are here{{ end }}
    2. {{- else }}
  2. {{ $n }} {{ $title }} coming
    3. {{- end }} {{- end }} {{- else }} {{- range $i, $p := $sorted }} {{- $n := add $i 1 }} {{- $isCurrent := eq $p.RelPermalink $current.RelPermalink }}
  3. {{ $n }} {{ if $p.Params.series_title }}{{ $p.Params.series_title }}{{ else }}{{ $p.Title }}{{ end }} {{- if $isCurrent }}you are here{{ end }}
    4. {{- end }} {{- end }}
{{- with $seriesPage }}

All parts of {{ $term }} →

{{- end }}
{{- end -}} {{- end -}}
