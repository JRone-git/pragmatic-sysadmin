Before this partial existed, only 1 of 36 articles linked to any page under /tools/, so the tool pages collected almost none of the site's internal link equity — even though they are the pages built to be shared and linked to. Tools and their tags come from catalog/tools.yaml (site.Data.tools), the same single source of truth that layouts/sitemap.xml and scripts/inject-tool-cta.py read. Matching is by shared tag against the site's real tag vocabulary, with a section-based fallback for posts that carry no matching tags. Rendered by layouts/partials/extend_post_content.html, between the article body and the newsletter CTA. */}} {{- $articleSections := slice "posts" "senior-tech" "sysadmin" "meta" -}} {{- $tools := site.Data.tools.tools -}} {{- if and $tools .IsPage (not .IsHome) (in $articleSections .Type) -}} {{- /* Normalise the post's tags so "DIY" and "diy" match the same tool. */ -}} {{- $pageTags := slice -}} {{- range .Params.tags -}} {{- $pageTags = $pageTags | append (lower .) -}} {{- end -}} {{- /* Score every tool by how many tags it shares with this post. */ -}} {{- $scored := slice -}} {{- range $tools -}} {{- $hits := 0 -}} {{- range .tags -}} {{- if in $pageTags (lower .) -}} {{- $hits = add $hits 1 -}} {{- end -}} {{- end -}} {{- if gt $hits 0 -}} {{- $scored = $scored | append (dict "hits" $hits "tool" .) -}} {{- end -}} {{- end -}} {{- /* No tag overlap? Fall back to tools aimed at this post's audience, so an untagged post still gets relevant links instead of none. */ -}} {{- $picked := slice -}} {{- if $scored -}} {{- $picked = first 3 (sort $scored "hits" "desc") -}} {{- else -}} {{- $audience := cond (eq .Type "senior-tech") "senior" "sysadmin" -}} {{- range first 3 (where $tools "audience" $audience) -}} {{- $picked = $picked | append (dict "hits" 0 "tool" .) -}} {{- end -}} {{- end -}} {{- with $picked }}
